program TC3;
const
  MAX == 5;
  PI == 3.14;

var
  counter: integer;
  scores: array[1..5] of integer;
  avg: real;
  name: string;
  found: boolean;

procedure Reset;
var
  i: integer;
begin
  i := 1;
  while i > 0 do
  begin
    counter := 0;
    i := 0;
  end;
end;

procedure SetScore(idx, val: integer);
begin
  scores[idx] := val;
end;

function Sum(n: integer): integer;
var
  i, total: integer;
begin
  total := 0;
  i := 1;
  for i := 1 to n do
  begin
    total := total + scores[i];
  end;
  Sum := total;
end;

begin
  counter := 0;
  found := false;
  Reset;
  SetScore(1, 90);
  SetScore(2, 85);
  avg := Sum(MAX);
  if avg > 80 then
    found := true;
end.
