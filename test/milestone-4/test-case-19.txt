program TC19;
var
  i : integer;
  y : integer;
begin
  { Edge 1: for TO, from = to (tepat 1 iterasi) }
  y := 0;
  for i := 5 to 5 do
  begin
    y := y + i
  end;
  writeln(y);

  { Edge 2: for DOWNTO, from = to (tepat 1 iterasi) }
  y := 0;
  for i := 3 downto 3 do
  begin
    y := y + i
  end;
  writeln(y);

  { Edge 3: for TO, iterasi banyak }
  y := 0;
  for i := 1 to 10 do
  begin
    y := y + i
  end;
  writeln(y);

  { Edge 4: for DOWNTO, iterasi banyak }
  y := 0;
  for i := 10 downto 1 do
  begin
    y := y + i
  end;
  writeln(y);

  { Edge 5: for dengan body yang lebih kompleks (if di dalam for) }
  y := 0;
  for i := 1 to 5 do
  begin
    if i > 2 then
      y := y + i
  end;
  writeln(y)
end.
