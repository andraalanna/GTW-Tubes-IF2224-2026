program TestRecord;
type
  Point == record
    x, y: integer
  end;
  Point3D == record
    p: Point;
    z: integer
  end;
var
  pt: Point;
  pt3d: Point3D;
begin
  pt.x := 42;
  pt.y := pt.x + 8;
  writeln(pt.x);
  writeln(pt.y);

  pt3d.p.x := 100;
  pt3d.p.y := pt3d.p.x * 2;
  pt3d.z := 300;
  writeln(pt3d.p.x);
  writeln(pt3d.p.y);
  writeln(pt3d.z);
end.
