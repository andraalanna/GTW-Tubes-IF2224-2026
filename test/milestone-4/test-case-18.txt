program TC18;
var
  x : integer;
  y : integer;
begin
  { Edge 1: kondisi langsung false, body tidak pernah dieksekusi }
  x := 10;
  y := 0;
  while x < 5 do
  begin
    y := y + 1;
    x := x + 1
  end;
  writeln(y);

  { Edge 2: loop hanya 1 iterasi }
  x := 4;
  y := 0;
  while x < 5 do
  begin
    y := y + 1;
    x := x + 1
  end;
  writeln(y);

  { Edge 3: nested while }
  x := 1;
  y := 0;
  while x <= 3 do
  begin
    y := y + x;
    x := x + 1
  end;
  writeln(y);

  { Edge 4: while dengan kondisi berubah di dalam body }
  x := 0;
  y := 1;
  while y < 100 do
  begin
    y := y * 2;
    x := x + 1
  end;
  writeln(x)
end.
